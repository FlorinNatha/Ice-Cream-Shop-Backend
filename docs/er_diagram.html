Delivergate Ice Cream Shop - Entity Relationship Diagram

Database structure implemented with Laravel 11/13 and MySQL 8.0

erDiagram
    USERS {
        bigint id PK
        string name
        string email UK
        string password
        enum role "owner | manager | cashier"
        timestamps timestamps
    }

    CATEGORIES {
        bigint id PK
        string name UK
        string slug UK
        text description
        timestamps timestamps
    }

    ITEMS {
        bigint id PK
        bigint category_id FK
        string name "INDEXED"
        string flavor
        decimal price
        int stock_quantity
        string unit
        text description
        timestamps timestamps
    }

    CUSTOMERS {
        bigint id PK
        string name
        string email UK
        string phone
        text address
        text notes
        timestamps timestamps
    }

    PERSONAL_ACCESS_TOKENS {
        bigint id PK
        string tokenable_type
        bigint tokenable_id
        string token UK
        timestamps timestamps
    }

    USERS ||--o{ PERSONAL_ACCESS_TOKENS : "has"
    CATEGORIES ||--o{ ITEMS : "has"
